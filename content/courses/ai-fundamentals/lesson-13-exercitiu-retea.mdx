### Pasul 1: Generăm datele

```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import make_moons
import torch
import torch.nn as nn

# Fixăm seed pentru reproductibilitate
np.random.seed(42)
torch.manual_seed(42)

# Generăm date în formă de "două lune" (non-linear, imposibil de separat cu o linie)
X_np, y_np = make_moons(n_samples=500, noise=0.2, random_state=42)

# Convertim la tensori PyTorch
X = torch.FloatTensor(X_np)
y = torch.FloatTensor(y_np).unsqueeze(1)  # [500] → [500, 1]

# Vizualizăm datele
plt.figure(figsize=(8, 6))
plt.scatter(X_np[y_np==0, 0], X_np[y_np==0, 1],
            c='royalblue', label='Clasa 0', alpha=0.7)
plt.scatter(X_np[y_np==1, 0], X_np[y_np==1, 1],
            c='tomato', label='Clasa 1', alpha=0.7)
plt.title("Datele noastre — Two Moons")
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()
```

**Ce observi?** Cele două clase sunt intercalate — nicio linie dreaptă nu le poate separa. Avem nevoie de o **decizie non-lineară**. Exact ce face rețeaua neuronală.

### Pasul 2: Definim rețeaua neuronală

```python
class MoonsClassifier(nn.Module):
    def __init__(self):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(2, 16),     # Input: 2 coordonate (x, y) → 16 neuroni
            nn.ReLU(),
            nn.Linear(16, 16),    # Strat ascuns: 16 → 16
            nn.ReLU(),
            nn.Linear(16, 1),     # Output: 1 valoare (probabilitate)
            nn.Sigmoid()          # → transformă în [0, 1]
        )

    def forward(self, x):
        return self.net(x)

model = MoonsClassifier()

# Câți parametri are rețeaua noastră?
total_params = sum(p.numel() for p in model.parameters())
print(f"Total parametri: {total_params}")
# Output: Total parametri: 593
# (2×16 + 16) + (16×16 + 16) + (16×1 + 1) = 32+16 + 256+16 + 16+1 = 337... + biasuri
```

### Pasul 3: Antrenăm modelul

```python
optimizer = torch.optim.Adam(model.parameters(), lr=0.01)
loss_fn = nn.BCELoss()  # Binary Cross-Entropy Loss pentru clasificare binară

losses = []  # Ținem evidența loss-ului pentru grafic

for epoca in range(1000):
    # Forward pass
    y_pred = model(X)
    loss = loss_fn(y_pred, y)

    # Backward pass
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

    losses.append(loss.item())

    if (epoca + 1) % 100 == 0:
        # Calculăm acuratețea
        with torch.no_grad():
            predictii = (model(X) > 0.5).float()
            acuratete = (predictii == y).float().mean()
        print(f"Epoca {epoca+1:4d} | Loss: {loss.item():.4f} | Acuratețe: {acuratete:.1%}")
```

**Output așteptat:**
```
Epoca  100 | Loss: 0.3821 | Acuratețe: 84.4%
Epoca  200 | Loss: 0.2134 | Acuratețe: 91.2%
Epoca  500 | Loss: 0.1023 | Acuratețe: 96.8%
Epoca 1000 | Loss: 0.0712 | Acuratețe: 97.4%
```

Observă cum acuratețea crește pe parcursul antrenării — exact arcașul nostru din Lecția 7 care se îmbunătățește treptat!

### Pasul 5: Vizualizăm granița de decizie

Acesta e cel mai interesant grafic — vom vedea cum "gândește" rețeaua:

```python
def plot_decision_boundary(model, X_np, y_np):
    # Creăm o grilă de puncte care acoperă spațiul de date
    x_min, x_max = X_np[:, 0].min() - 0.5, X_np[:, 0].max() + 0.5
    y_min, y_max = X_np[:, 1].min() - 0.5, X_np[:, 1].max() + 0.5

    xx, yy = np.meshgrid(np.linspace(x_min, x_max, 300),
                          np.linspace(y_min, y_max, 300))

    # Trecem toată grila prin model
    grid_tensor = torch.FloatTensor(np.c_[xx.ravel(), yy.ravel()])
    with torch.no_grad():
        Z = model(grid_tensor).numpy().reshape(xx.shape)

    # Plotăm
    plt.figure(figsize=(9, 7))
    plt.contourf(xx, yy, Z, alpha=0.3,
                 cmap=plt.cm.RdBu, levels=np.linspace(0, 1, 11))
    plt.colorbar(label='Probabilitate Clasa 1')

    plt.scatter(X_np[y_np==0, 0], X_np[y_np==0, 1],
                c='royalblue', label='Clasa 0', edgecolors='k', s=50)
    plt.scatter(X_np[y_np==1, 0], X_np[y_np==1, 1],
                c='tomato', label='Clasa 1', edgecolors='k', s=50)

    plt.title("Granița de decizie a rețelei neuronale")
    plt.legend()
    plt.grid(True, alpha=0.2)
    plt.show()

plot_decision_boundary(model, X_np, y_np)
```
